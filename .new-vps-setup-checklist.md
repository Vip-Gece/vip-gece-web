### 01.4 Swap Alanı (2 GB RAM veya altı için)

```bash
# RAM kontrolü
free -h

# Eğer RAM < 2 GB ise swap oluştur
if [ $(free -m | awk '/^Mem:/{print $2}') -lt 2000 ]; then
  fallocate -l 2G /swapfile
  chmod 600 /swapfile
  mkswap /swapfile
  swapon /swapfile
  echo '/swapfile none swap sw 0 0' >> /etc/fstab
fi
```

### 01.5 Sysctl Hardening

```bash
cat > /etc/sysctl.d/99-vip-gece-hardening.conf << 'EOF'
# IP spoofing koruması
net.ipv4.conf.all.rp_filter = 1
net.ipv4.conf.default.rp_filter = 1

# ICMP redirect devre dışı
net.ipv4.conf.all.accept_redirects = 0
net.ipv4.conf.default.accept_redirects = 0
net.ipv4.conf.all.send_redirects = 0

# Source routing devre dışı
net.ipv4.conf.all.accept_source_route = 0
net.ipv4.conf.default.accept_source_route = 0

# SYN flood koruması
net.ipv4.tcp_syncookies = 1
net.ipv4.tcp_max_syn_backlog = 2048

# TCP bağlantı zaman aşımı
net.ipv4.tcp_fin_timeout = 15

# Port aralığı
net.ipv4.ip_local_port_range = 1024 65535

# File descriptor limiti
fs.file-max = 2097152
EOF

sysctl --system
```

## 02 — KULLANICI VE SSH

### 02.1 Ubuntu Kullanıcısı Oluştur

```bash
# ubuntu kullanıcısı oluştur (yoksa)
useradd -m -s /bin/bash ubuntu

# sudo yetkisi ver
usermod -aG sudo ubuntu

# sudo şifre istemesini kaldır (opsiyonel, kolaylık için)
echo "ubuntu ALL=(ALL) NOPASSWD:ALL" > /etc/sudoers.d/ubuntu
chmod 440 /etc/sudoers.d/ubuntu
```

### 02.2 SSH Anahtarı Ekle

```bash
# ubuntu kullanıcısına geç
su - ubuntu

# .ssh dizini oluştur
mkdir -p ~/.ssh
chmod 700 ~/.ssh

# Public key ekle (yerel Mac'ten kopyaladığın)
nano ~/.ssh/authorized_keys
# Public key'i yapıştır, kaydet

# Yetkileri ayarla
chmod 600 ~/.ssh/authorized_keys

# root'a dön
exit
```

### 02.3 SSH Hardening

```bash
# SSH config'i düzenle
nano /etc/ssh/sshd_config

# Değiştir/ekle:
Port 22
PermitRootLogin no
PasswordAuthentication no
PubkeyAuthentication yes
ChallengeResponseAuthentication no
UsePAM yes
X11Forwarding no
PrintMotd no
AcceptEnv LANG LC_*
Subsystem sftp /usr/lib/openssh/sftp-server
MaxAuthTries 3
LoginGraceTime 30
ClientAliveInterval 300
ClientAliveCountMax 2

# SSH servisini yeniden başlat
systemctl restart sshd
```

## 03 — FIREWALL (UFW)

### 03.1 UFW Kurulumu

```bash
# UFW kur
apt install ufw -y

# Varsayılan politikalar
ufw default deny incoming
ufw default allow outgoing

# SSH'e izin ver
ufw allow ssh
ufw allow 22/tcp

# HTTP ve HTTPS'e izin ver
ufw allow http
ufw allow https
ufw allow 80/tcp
ufw allow 443/tcp

# UFW'u etkinleştir (EVET de)
ufw enable

# Durumu kontrol et
ufw status verbose
```

### 03.2 Fail2Ban (Opsiyonel, SSH için varsayılan kapalı)

```bash
apt install fail2ban -y

# SSH jail'i varsayılan kapalı tut.
# Not: Sunucuya erişim doğrulanmadan veya güvenilir yönetim IP'leri netleşmeden
# SSH fail2ban açılırsa doğru anahtar kullanılsa bile acil erişimi kesebilir.
cat > /etc/fail2ban/jail.local << 'EOF'
[sshd]
enabled = false
port = ssh
filter = sshd
logpath = /var/log/auth.log
maxretry = 3
bantime = 3600
findtime = 600
EOF

# Servisi otomatik başlatma; SSH erişim testi ve allowlist tamamlanınca bilinçli aç.
systemctl disable --now fail2ban || true
```

### 09.2 vip-gece.site.conf

```bash
sudo nano /etc/nginx/sites-available/vip-gece.site.conf
```

```nginx
map $http_upgrade $vip_gece_site_connection_upgrade {
    default upgrade;
    '' close;
}

server {
    listen 80;
    listen [::]:80;
    server_name vip-gece.site www.vip-gece.site;

    access_log /var/log/nginx/vip-gece.site.access.log;
    error_log /var/log/nginx/vip-gece.site.error.log;
    real_ip_header X-Vip-Gece-Origin-Peer;
    real_ip_recursive off;
    include /etc/nginx/snippets/vip-gece-cloudflare-only.conf;

    location ^~ /.well-known/acme-challenge/ {
        root /var/www/html;
        allow all;
        try_files $uri =404;
    }

    location / {
        return 301 https://vip-gece.site$request_uri;
    }
}

server {
    listen 443 ssl;
    listen [::]:443 ssl;
    server_name vip-gece.site www.vip-gece.site;

    access_log /var/log/nginx/vip-gece.site.access.log;
    error_log /var/log/nginx/vip-gece.site.error.log;
    real_ip_header X-Vip-Gece-Origin-Peer;
    real_ip_recursive off;
    include /etc/nginx/snippets/vip-gece-cloudflare-only.conf;
    client_max_body_size 16m;

    ssl_certificate /etc/letsencrypt/live/vip-gece.site/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/vip-gece.site/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    if ($host = www.vip-gece.site) {
        return 301 https://vip-gece.site$request_uri;
    }

    location ^~ /.well-known/acme-challenge/ {
        root /var/www/html;
        allow all;
        try_files $uri =404;
    }

    location / {
        proxy_pass http://127.0.0.1:3003;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $http_cf_connecting_ip;
        proxy_set_header X-Forwarded-For $http_cf_connecting_ip;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Port $server_port;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection $vip_gece_site_connection_upgrade;
        proxy_read_timeout 60s;
        proxy_send_timeout 60s;

        proxy_hide_header X-Powered-By;
        proxy_hide_header Strict-Transport-Security;
        proxy_hide_header X-Content-Type-Options;
        proxy_hide_header Referrer-Policy;
        proxy_hide_header X-Frame-Options;
        proxy_hide_header Permissions-Policy;
    }

    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;
    add_header X-Frame-Options "DENY" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;
    add_header Permissions-Policy "geolocation=(), microphone=(), camera=(), payment=(), usb=()" always;
}
```
